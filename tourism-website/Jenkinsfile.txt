
pipeline {
    agent { label 'york' }

    options {
        timestamps()
    }

    environment {
        AWS_REGION = 'ap-south-1'
        SECRET_ID  = 'tourism/dockerhub'
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Get Docker Hub Usernames') {
            steps {
                script {
                    def secretJson = sh(
                        script: '''
                            set +x
                            aws secretsmanager get-secret-value \
                              --secret-id "$SECRET_ID" \
                              --region "$AWS_REGION" \
                              --query SecretString \
                              --output text
                        ''',
                        returnStdout: true
                    ).trim()

                    def secret = readJSON text: secretJson

                    if (!secret.username || !secret.password) {
                        error('Docker Hub username or password is missing from Secrets Manager')
                    }

                    env.DOCKERHUB_USERNAME = secret.username
                    env.IMAGE_NAME = "${secret.username}/tourism-website"
                    env.IMAGE_TAG = "${env.BUILD_NUMBER}"

                    secretJson = null
                    secret = null
                }
            }
        }

        stage('Build') {
            steps {
                sh '''
                    set -eu
                    docker build -t "$IMAGE_NAME:$IMAGE_TAG" .
                '''
            }
        }

        stage('Push to Docker Hub') {
            steps {
                sh '''
                    set +x
                    set -eu
                    umask 077

                    export DOCKER_CONFIG=$(mktemp -d)
                    trap 'docker logout >/dev/null 2>&1 || true; rm -rf "$DOCKER_CONFIG"' EXIT

                    SECRET_JSON=$(aws secretsmanager get-secret-value \
                      --secret-id "$SECRET_ID" \
                      --region "$AWS_REGION" \
                      --query SecretString \
                      --output text)

                    DOCKER_USER=$(printf '%s' "$SECRET_JSON" | jq -er '.username')
                    DOCKER_TOKEN=$(printf '%s' "$SECRET_JSON" | jq -er '.password')
                    unset SECRET_JSON

                    printf '%s' "$DOCKER_TOKEN" | docker login \
                      --username "$DOCKER_USER" \
                      --password-stdin

                    unset DOCKER_TOKEN DOCKER_USER

                    docker push "$IMAGE_NAME:$IMAGE_TAG"
                '''
            }
        }

        stage('Deploy Staging') {
            environment {
                HOST_PORT = '8081'
            }
            steps {
                sh '''
                    set -eu
                    docker rm -f tourism-staging 2>/dev/null || true

                    docker run -d \
                      --restart unless-stopped \
                      --name tourism-staging \
                      -p "$HOST_PORT:80" \
                      "$IMAGE_NAME:$IMAGE_TAG"
                '''
                echo 'Staging deployment completed'
            }
        }

        stage('Test Staging') {
            steps {
                sh '''
                    set -eu
                    curl --retry 5 --retry-delay 2 --retry-connrefused \
                      --fail http://localhost:8081/
                '''
            }
        }

        stage('Approve Production') {
            steps {
                input message: 'Has Staging been tested successfully?',
                      ok: 'Deploy Production'
            }
        }

        stage('Deploy Production') {
            environment {
                HOST_PORT = '8082'
            }
            steps {
                sh '''
                    set -eu
                    docker rm -f tourism-production 2>/dev/null || true

                    docker run -d \
                      --restart unless-stopped \
                      --name tourism-production \
                      -p "$HOST_PORT:80" \
                      "$IMAGE_NAME:$IMAGE_TAG"
                '''
                echo 'Production deployment completed'
            }
        }
    }

    post {
        success {
            echo 'Tourism CI/CD pipeline completed successfully'
        }
        failure {
            echo 'Tourism CI/CD pipeline failed'
        }
    }
}
